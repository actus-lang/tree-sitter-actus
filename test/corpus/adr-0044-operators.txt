==================
ADR-0044 operators preserve compiler precedence
==================

verb main() -> Bool {
  erg left: u8 = 13;
  erg right: u8 = 5;
  erg count: u8 = 1;
  return left % right == 3 && (left & right) ^ left != 0 || left << count >= right;
}
